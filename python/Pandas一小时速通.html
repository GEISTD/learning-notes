10 minSeries 与 DataFrame

DataFrame 就是一张表格（有行有列），Series 是表格里的一列。

import pandas as pd

# Series = 一列数据（类似带索引的 list）
s = pd.Series([10, 20, 30, 40])
print(s)
# 输出：
# 0    10
# 1    20
# 2    30
# 3    40
# dtype: int64

# DataFrame = 一张表（类似 Excel / dict 的集合）
df = pd.DataFrame({
    'name':   ['张三', '李四', '王五', '赵六'],
    'age':    [20, 21, 19, 22],
    'score':   [85, 92, 78, 90],
    'gender': ['男', '男', '女', '女']
})
print(df)
#    name  age  score gender
# 0  张三   20     85     男
# 1  李四   21     92     男
# 2  王五   19     78     女
# 3  赵六   22     90     女

# 核心属性
print(df.shape)        # (4, 4) → 4行4列
print(df.columns)      # 列名列表
print(df.index)        # 行索引（默认0,1,2,3）
print(df.dtypes)        # 每列的数据类型
print(df.head(2))       # 前两行（head默认5行）
print(df.info())        # 整体信息概览（类型+非空数+内存）
print(df.describe())    # 数值列的统计摘要（均值/标准差/最值/分位数）

15 min筛选与定位
# === 选列 ===
df['name']           # 选一列 → 返回 Series
df[['name', 'age']]  # 选多列 → 返回 DataFrame

# === 选行 ===
df[0:2]               # 切片：前两行
df.iloc[0]              # 按位置取第0行 → Series
df.iloc[0:3]           # 按位置取0,1,2行
df.loc[0]              # 按标签取行（默认索引也是0,1,2...）

# === 条件筛选（最重要！） ===
df[df['age'] > 20]                      # age>20 的所有行
df[df['gender'] == '男']                   # 性别为男的行
df[(df['age'] > 20) & (df['score'] >= 85)]  # 多条件：且（用&）
df[(df['gender'] == '男') | (df['score'] > 85)]   # 多条件：或（用|）

# === 常用方法 ===
df.sort_values('score', ascending=False)  # 按分数降序排
df.sort_values(['gender', 'score'], ascending=[True, False])  # 先按性别再按分数
df['score'].max()       # 最高分
df['score'].mean()      # 平均分
df['score'].idxmax()    # 最高分所在行的索引
df.nlargest(2, 'score')   # 分数前2名
df['name'].unique()     # 唯一值列表
df['name'].value_counts() # 每个值出现次数
练习：从上面的 df 中找出：①女生中分数最高的；②年龄大于19且分数大于80的人的name。

10 min缺失值处理
# 造一份含缺失值的脏数据
import numpy as np
dirty = pd.DataFrame({
    'name':   ['张三', '李四', None, '赵六', '钱七'],
    'age':    [20, np.nan, 19, 22, 21],
    'score':   [85, 92, 78, np.nan, 90],
    'gender': ['男', '男', '女', '女', None]
})

# 检查缺失值
dirty.isnull()          # 每个格子是否为NaN（True/False表）
dirty.isnull().sum()    # 每列有多少个缺失值（最常用！）
dirty.info()            # 也能看到每列非空数

# 删除缺失值
dirty.dropna()              # 删掉任何含NaN的行
dirty.dropna(subset=['age'])   # 只删age列缺失的行
dirty.dropna(thresh=3)       # 至少3个非空值才保留该行

# 填充缺失值
dirty['age'].fillna(dirty['age'].mean())   # 用平均年龄填充
dirty['score'].fillna(0)                # 用0填充分数
dirty['name'].fillna('未知')               # 用字符串填充
dirty['gender'].fillna(dirty['gender'].mode()[0])  # 用众数填充

# 填充后要赋值回去（Pandas操作默认不修改原数据）
dirty['age'] = dirty['age'].fillna(dirty['age'].mean())
易错点：Pandas 的大多数操作（fillna、dropna、sort_values）都不会修改原数据，而是返回新对象。要么赋值回去 df['x'] = df['x'].fillna(...)，要么用 inplace=True 参数。
10 min综合练习：清洗一份脏数据
# 这是一份模拟的"学生成绩脏数据"——直接复制运行
import pandas as pd
import numpy as np

dirty_data = pd.DataFrame({
    '姓名': ['张三', '李四', None, '赵六', '钱七', '孙八', '周九', '吴十', '郑十一', '王十二'],
    '年龄': [20, 21, 19, np.nan, 22, 20, 21, np.nan, 19, 23],
    '成绩': [85, 92, 78, 90, np.nan, 88, 95, 76, 82, np.nan],
    '性别': ['男', '男', '女', '女', '男', None, '女', '男', '女', '男'],
    '班级': ['A班', 'B班', 'A班', 'B班', 'A班', 'B班', 'A班', 'B班', 'A班', 'B班']
})
任务（限时10分钟，不用AI）：
  1. 查看每列有多少缺失值
  2. 把「姓名」缺失的行删掉
  3. 把「年龄」缺失值用平均年龄填充
  4. 把「成绩」缺失值用0填充
  5. 把「性别」缺失值用众数填充
  6. 按班级分组，算出每班的平均成绩和人数
  7. 按成绩降序排序，打印前3名
  8. 把清洗后的数据导出为 clean_students.csv
做完后对照参考答案 
# 1. 查看缺失值
print(dirty_data.isnull().sum())

# 2. 删除姓名缺失的行
dirty_data = dirty_data.dropna(subset=['姓名'])

# 3. 年龄用均值填充
dirty_data['年龄'] = dirty_data['年龄'].fillna(dirty_data['年龄'].mean())

# 4. 成绩用0填充
dirty_data['成绩'] = dirty_data['成绩'].fillna(0)

# 5. 性别用众数填充
dirty_data['性别'] = dirty_data['性别'].fillna(dirty_data['性别'].mode()[0])

# 6. 按班级分组统计
class_stats = dirty_data.groupby('班级').agg({
    '成绩': 'mean',
    '姓名': 'count'
}).reset_index()
print(class_stats)

# 7. 成绩前3名
top3 = dirty_data.nlargest(3, '成绩')
print(top3[['姓名', '成绩']])

# 8. 导出
dirty_data.to_csv('clean_students.csv', index=False, encoding='utf-8-sig')
print('清洗完成，已保存 clean_students.csv')
