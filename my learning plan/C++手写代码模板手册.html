C++ 手写代码模板手册

数据结构基础模板 + 常见扩展模式 | 每个模板默写5遍到肌肉记忆
使用方法：每天选3-5个模板，关掉所有资料，纸笔默写。写到不看代码能完整写出来为止。每个模板下面有"默写检查点"，全部满足才算过关。
一、数组模板
1.1 一维数组遍历（正序/逆序）基础
// 正序遍历
for (int i = 0; i < n; i++) {
    // a[i]
}

// 逆序遍历
for (int i = n - 1; i >= 0; i--) {
    // a[i]
}

// range-based for（只读）
for (int x : arr) {
    // x 是副本
}

// range-based for（引用，可修改）
for (int& x : arr) {
    x *= 2;
}
默写检查点：①逆序循环i的初始值是n-1不是n ②range-based for加&才能修改原数组

    // matrix[i][n-1-i] 是副对角线元素
}

// 矩阵转置
for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
        swap(matrix[i][j], matrix[j][i]);
    }
}

1.3 双指针（对撞/快慢）高频
// 对撞指针（两端向中间）— 回文判断、两数之和
int left = 0, right = n - 1;
while (left < right) {
    if (arr[left] + arr[right] == target) break;
    else if (arr[left] + arr[right] < target) left++;
    else right--;
}

// 快慢指针（同端出发）— 删除重复元素、找中点
int slow = 0;
for (int fast = 0; fast < n; fast++) {
    if (fast == 0 || arr[fast] != arr[fast - 1]) {
        arr[slow++] = arr[fast];
    }
}
// slow 就是去重后的长度

// 滑动窗口 — 最小覆盖子串框架
int left = 0, right = 0;
while (right < n) {
    // 扩大右边界：将 arr[right] 加入窗口
    right++;
    while (窗口满足条件) {
        // 更新答案
        // 缩小左边界：将 arr[left] 移出窗口
        left++;
    }
}
默写检查点：①对撞指针while条件是left<right不是left<=right ②快慢指针的fast走快slow走慢 ③滑动窗口外层扩右、内层缩左

2.1 链表节点定义与遍历基础

struct ListNode {

    ListNode* next;

// 遍历链表（核心模板，写到烂熟）
ListNode* cur = head;
while (cur != nullptr) {
    // 访问 cur->val
    cur = cur->next;
}

// 统计链表长度
int len(ListNode* head) {
    int n = 0;
    ListNode* cur = head;
    while (cur) { n++; cur = cur->next; }
    return n;
}
默写检查点：①构造函数初始化列表写法 ②遍历while条件是cur!=nullptr ③cur=cur->next必须在循环体内
2.2 链表逆置（四步法）必背
// 链表逆置 — 四步法（面试最高频之一）
ListNode* reverseList(ListNode* head) {
    ListNode* prev = nullptr;
    ListNode* cur = head;
    while (cur != nullptr) {
        ListNode* next = cur->next;  // 1. 先存下一个
        cur->next = prev;            // 2. 反转指向
        prev = cur;                  // 3. prev前进
        cur = next;                  // 4. cur前进
    }
    return prev;  // prev 是新的头
}

// 区间逆置 [left, right]（头插法）
ListNode* reverseBetween(ListNode* head, int left, int right) {
    ListNode* dummy = new ListNode(0);
    dummy->next = head;
    ListNode* prev = dummy;
    for (int i = 0; i < left - 1; i++) prev = prev->next;
    ListNode* cur = prev->next;
    for (int i = 0; i < right - left; i++) {
        ListNode* next = cur->next;
        cur->next = next->next;
        next->next = prev->next;
        prev->next = next;
    }
    return dummy->next;
}
默写检查点：①四步顺序：存next→反转→prev前进→cur前进 ②返回prev不是cur ③区间逆置用dummy节点防止头节点变化
2.3 链表常用扩展扩展
// 快慢指针找中点（判断环/分割链表）
ListNode* slow = head, *fast = head;
while (fast && fast->next) {
    slow = slow->next;
    fast = fast->next->next;
}
// slow 指向中点（奇数个节点时是中间，偶数个时是前半段最后一个）

// 判断环形链表
bool hasCycle(ListNode* head) {
    ListNode* slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;
    }
    return false;
}

// 合并两个有序链表
ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
    ListNode* dummy = new ListNode(0);
    ListNode* tail = dummy;
    while (l1 && l2) {
        if (l1->val <= l2->val) { tail->next = l1; l1 = l1->next; }
        else { tail->next = l2; l2 = l2->next; }
        tail = tail->next;
    }
    tail->next = l1 ? l1 : l2;  // 接上剩余部分
    return dummy->next;
}

// 删除倒数第N个节点（双指针一前一后）
ListNode* removeNthFromEnd(ListNode* head, int n) {
    ListNode* dummy = new ListNode(0);
    dummy->next = head;
    ListNode* fast = dummy, *slow = dummy;
    for (int i = 0; i < n; i++) fast = fast->next;
    while (fast->next) { fast = fast->next; slow = slow->next; }
    ListNode* del = slow->next;
    slow->next = del->next;
    delete del;
    return dummy->next;
}
默写检查点：①快慢指针找中点while条件是fast&&fast->next ②合并链表用dummy+tail ③删除节点记得delete

stack<int> st;
st.push(x);     // 入栈

st.pop();         // 弹出栈顶

// 括号匹配（经典栈应用）
bool isValid(string s) {
    stack<char> st;
    for (char c : s) {
        if (c == '(' || c == '[' || c == '{')
            st.push(c);
        else {
            if (st.empty()) return false;
            char top = st.top();
            if ((c == ')' && top != '(') ||
                (c == ']' && top != '[') ||
                (c == '}' && top != '{'))
                return false;
            st.pop();
        }
    }
    return st.empty();
}

// 单调栈 — 每日温度（找右边第一个更大元素）
vector<int> dailyTemperatures(vector<int>& T) {
    int n = T.size();
    vector<int> ans(n, 0);
    stack<int> st;  // 存下标
    for (int i = 0; i < n; i++) {
        while (!st.empty() && T[i] > T[st.top()]) {
            ans[st.top()] = i - st.top();
            st.pop();
        }
        st.push(i);
    }
    return ans;
}
四、二叉树模板
4.1 二叉树定义与DFS遍历必背
// 节点定义
struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

// 递归三部曲：①base case ②左子树递归 ③右子树递归 ④利用左右子树的返回值

// 前序遍历（根→左→右）
void preorder(TreeNode* root) {
    if (!root) return;       // base case
    cout << root->val;       // 访问根
    preorder(root->left);   // 左
    preorder(root->right);  // 右
}

// 中序遍历（左→根→右）— BST中序遍历是有序的
void inorder(TreeNode* root) {
    if (!root) return;
    inorder(root->left);
    cout << root->val;
    inorder(root->right);
}

// 后序遍历（左→右→根）— 求高度必须后序
void postorder(TreeNode* root) {
    if (!root) return;
    postorder(root->left);
    postorder(root->right);
    cout << root->val;
}

// 求最大深度（后序的典型应用）
int maxDepth(TreeNode* root) {
    if (!root) return 0;
    int left = maxDepth(root->left);
    int right = maxDepth(root->right);
    return max(left, right) + 1;
}
默写检查点：①base case是if(!root) return ②前/中/后序的区别只是"访问根"那行代码的位置 ③求深度=后序+max(左,右)+1
4.2 BFS层序遍历必背
// BFS层序遍历（队列实现）— 核心模板
vector<vector<int>> levelOrder(TreeNode* root) {
    vector<vector<int>> result;
    if (!root) return result;

    queue<TreeNode*> q;
    q.push(root);

    while (!q.empty()) {
        int levelSize = q.size();  // 当前层节点数
        vector<int> level;

        for (int i = 0; i < levelSize; i++) {
            TreeNode* node = q.front();
            q.pop();
            level.push_back(node->val);

            if (node->left)  q.push(node->left);
            if (node->right) q.push(node->right);
        }
        result.push_back(level);
    }
    return result;
}

// BFS的固定四步：
// 1. 创建队列，push初始节点
// 2. while队列不空
// 3. 记录当前层size，for循环处理这一层
// 4. 取出front→pop→处理→push子节点
默写检查点：①先记录levelSize再进for循环（因为队列在变）②pop在front之后 ③push子节点前判空
4.3 二叉树扩展扩展
// 判断对称二叉树
bool isSymmetric(TreeNode* root) {
    return check(root->left, root->right);
}
bool check(TreeNode* p, TreeNode* q) {
    if (!p && !q) return true;
    if (!p || !q) return false;
    return p->val == q->val
        && check(p->left, q->right)
        && check(p->right, q->left);
}

// 翻转二叉树
TreeNode* invertTree(TreeNode* root) {
    if (!root) return nullptr;
    swap(root->left, root->right);
    invertTree(root->left);
    invertTree(root->right);
    return root;
}

// 路径总和（DFS回溯思路）
bool hasPathSum(TreeNode* root, int target) {
    if (!root) return false;
    if (!root->left && !root->right)
        return root->val == target;
    return hasPathSum(root->left, target - root->val)
        || hasPathSum(root->right, target - root->val);
}
默写检查点：①对称判断是left.right vs right.left ②翻转就是swap左右再递归 ③路径和用减法不用加法
五、排序模板
5.1 快速排序必背
// 快排 — 分治三步：选基准→分区→递归
void quickSort(vector<int>& a, int lo, int hi) {
    if (lo >= hi) return;  // base case
    int pivot = a[lo + (hi - lo) / 2];
    int i = lo, j = hi;
    while (i <= j) {
        while (a[i] < pivot) i++;
        while (a[j] > pivot) j--;
        if (i <= j) {
            swap(a[i], a[j]);
            i++; j--;
        }
    }
    quickSort(a, lo, j);
    quickSort(a, i, hi);
}
5.2 归并排序必背
// 归并 — 分治三步：分半→递归排序→合并
void mergeSort(vector<int>& a, int lo, int hi) {
    if (lo >= hi) return;
    int mid = lo + (hi - lo) / 2;
    mergeSort(a, lo, mid);
    mergeSort(a, mid + 1, hi);

    // 合并两个有序段
    vector<int> tmp;
    int i = lo, j = mid + 1;
    while (i <= mid && j <= hi) {
        if (a[i] <= a[j]) tmp.push_back(a[i++]);
        else tmp.push_back(a[j++]);
    }
    while (i <= mid) tmp.push_back(a[i++]);
    while (j <= hi) tmp.push_back(a[j++]);
    for (int k = 0; k < tmp.size(); k++)
        a[lo + k] = tmp[k];
}
默写检查点：①快排的while(i<=j)条件含等号 ②快排递归调用是(lo,j)和(i,hi)不是(lo,mid)和(mid+1,hi) ③归并的合并部分用双指针比较大小 ④mid=lo+(hi-lo)/2防止溢出
六、二分查找模板
6.1 二分查找三套模板必背
// 模板1：精确查找（while条件 <=）
int binarySearch(vector<int>& a, int target) {
    int left = 0, right = a.size() - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (a[mid] == target) return mid;
        else if (a[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

// 模板2：找左边界（第一个 >= target 的位置）
int lowerBound(vector<int>& a, int target) {
    int left = 0, right = a.size();
    while (left < right) {
        int mid = left + (right - left) / 2;
        if (a[mid] < target) left = mid + 1;
        else right = mid;  // 注意不-1
    }
    return left;
}

// 模板3：找右边界（最后一个 <= target 的位置）
int upperBound(vector<int>& a, int target) {
    int left = 0, right = a.size();
    while (left < right) {
        int mid = left + (right - left) / 2;
        if (a[mid] <= target) left = mid + 1;
        else right = mid;
    }
    return left - 1;  // 右边界 = left - 1
}
默写检查点：①精确查找用left<=right+left=mid+1/right=mid-1 ②找左边界用left<right+right=mid ③找右边界返回left-1 ④mid永远用left+(right-left)/2防溢出

// 两数之和（哈希表经典应用）
vector<int> twoSum(vector<int>& nums, int target) {
    unordered_map<int, int> mp;  // val -> index
    for (int i = 0; i < nums.size(); i++) {
        int complement = target - nums[i];
        if (mp.count(complement))
            return {mp[complement], i};
        mp[nums[i]] = i;
    }
    return {};
}

八、回溯模板
8.1 回溯万能模板必背
// 回溯 = DFS + 选择 + 撤销选择
void backtrack(/* 路径, 选择列表 */) {
    if (满足结束条件) {
        // 记录结果
        return;
    }
    for (auto& choice : 选择列表) {
        // 做选择
        backtrack(/* 新路径, 新选择列表 */);
        // 撤销选择（回溯）
    }
}

// 全排列（经典回溯）
vector<vector<int>> result;
vector<int> path;
vector<bool> used;

void permute(vector<int>& nums) {
    used.assign(nums.size(), false);
    dfs(nums);
}
void dfs(vector<int>& nums) {
    if (path.size() == nums.size()) {
        result.push_back(path);
        return;
    }
    for (int i = 0; i < nums.size(); i++) {
        if (used[i]) continue;
        used[i] = true;           // 选择
        path.push_back(nums[i]);
        dfs(nums);
        path.pop_back();            // 撤销
        used[i] = false;
    }
}
默写检查点：①结束条件在最前 ②做选择→递归→撤销选择三步顺序 ③pop_back和used=false缺一不可
九、动态规划模板
9.1 一维DP进阶
// 爬楼梯 — 最简单的DP
// 状态：dp[i] = 爬到第i阶的方法数
// 转移：dp[i] = dp[i-1] + dp[i-2]
int climbStairs(int n) {
    if (n <= 2) return n;
    vector<int> dp(n + 1);
    dp[1] = 1;
    dp[2] = 2;
    for (int i = 3; i <= n; i++)
        dp[i] = dp[i - 1] + dp[i - 2];
    return dp[n];
}

// 最大子数组和（Kadane算法）
// 状态：dp[i] = 以i结尾的最大子数组和
// 转移：dp[i] = max(dp[i-1]+a[i], a[i])
int maxSubArray(vector<int>& a) {
    int maxSum = a[0], curSum = a[0];
    for (int i = 1; i < a.size(); i++) {
        curSum = max(curSum + a[i], a[i]);
        maxSum = max(maxSum, curSum);
    }
    return maxSum;
}

// 打家劫舍
// 状态：dp[i] = 前i家能偷的最大金额
// 转移：dp[i] = max(dp[i-1], dp[i-2]+nums[i])
int rob(vector<int>& nums) {
    int n = nums.size();
    if (n == 0) return 0;
    if (n == 1) return nums[0];
    vector<int> dp(n);
    dp[0] = nums[0];
    dp[1] = max(nums[0], nums[1]);
    for (int i = 2; i < n; i++)
        dp[i] = max(dp[i - 1], dp[i - 2] + nums[i]);
    return dp[n - 1];
}
默写检查点：①DP三步：定义状态→写转移方程→确定初始值 ②爬楼梯dp[1]=1,dp[2]=2 ③打家劫舍转移是max(不偷=dp[i-1], 偷=dp[i-2]+nums[i])
9.2 二维DP进阶
// 最小路径和
// 状态：dp[i][j] = 从(0,0)到(i,j)的最小路径和
// 转移：dp[i][j] = min(dp[i-1][j], dp[i][j-1]) + grid[i][j]
int minPathSum(vector<vector<int>>& grid) {
    int m = grid.size(), n = grid[0].size();
    vector<vector<int>> dp(m, vector<int>(n));
    dp[0][0] = grid[0][0];
    for (int i = 1; i < m; i++) dp[i][0] = dp[i - 1][0] + grid[i][0];
    for (int j = 1; j < n; j++) dp[0][j] = dp[0][j - 1] + grid[0][j];
    for (int i = 1; i < m; i++)
        for (int j = 1; j < n; j++)
            dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j];
    return dp[m - 1][n - 1];
}

// 最长公共子序列（LCS）
// 状态：dp[i][j] = s1[0..i-1]和s2[0..j-1]的LCS长度
// 转移：s1[i-1]==s2[j-1] ? dp[i-1][j-1]+1 : max(dp[i-1][j], dp[i][j-1])
int longestCommonSubsequence(string s1, string s2) {
    int m = s1.size(), n = s2.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    for (int i = 1; i <= m; i++)
        for (int j = 1; j <= n; j++)
            if (s1[i - 1] == s2[j - 1])
                dp[i][j] = dp[i - 1][j - 1] + 1;
            else
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
    return dp[m][n];
}
默写检查点：①二维DP先处理第一行第一列 ②LCS的dp下标和字符串下标差1（dp[1][1]对应s1[0]和s2[0]）

// 邻接表表示
vector<vector<int>> buildGraph(int n, vector<vector<int>>& edges) {
    vector<vector<int>> graph(n);
    for (auto& e : edges) {
        graph[e[0].push_back(e[1]);
        graph[e[1].push_back(e[0]);  // 无向图加这行
    }
    return graph;
}

// BFS遍历图（从start出发）
void bfs(vector<vector<int>>& graph, int start) {
    vector<bool> visited(graph.size(), false);
    queue<int> q;
    q.push(start);
    visited[start] = true;

    while (!q.empty()) {
        int cur = q.front();
        q.pop();
        for (int next : graph[cur]) {
            if (!visited[next]) {
                visited[next] = true;
                q.push(next);
            }
        }
    }
}

// DFS遍历图
vector<bool> visited;
void dfs(vector<vector<int>>& graph, int cur) {
    visited[cur] = true;
    for (int next : graph[cur]) {
        if (!visited[next])
            dfs(graph, next);
    }
}

// 岛屿数量（矩阵DFS经典题）
int numIslands(vector<vector<char>>& grid) {
    int count = 0;
    int m = grid.size(), n = grid[0].size();
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            if (grid[i][j] == '1') {
                dfsIsland(grid, i, j);
                count++;
            }
        }
    }
    return count;
}
void dfsIsland(vector<vector<char>>& grid, int i, int j) {
    if (i < 0 || i >= grid.size() || j < 0 || j >= grid[0].size()) return;
    if (grid[i][j] != '1') return;
    grid[i][j] = '0';  // 标记为已访问
    dfsIsland(grid, i + 1, j);
    dfsIsland(grid, i - 1, j);
    dfsIsland(grid, i, j + 1);
    dfsIsland(grid, i, j - 1);
}
默写检查点：①BFS入队时就标记visited（不是出队时）②岛屿DFS四个方向递归 ③边界检查在DFS最前面

// 生产者-消费者（condition_variable）
queue<int> taskQueue;
mutex mtx;
condition_variable cv;

// 生产者
void producer() {
    while (true) {
        int task = produce();
        {
            unique_lock<mutex> lock(mtx);
            taskQueue.push(task);
        }
        cv.notify_one();  // 唤醒一个消费者
    }
}

// 消费者
void consumer() {
    while (true) {
        unique_lock<mutex> lock(mtx);
        cv.wait(lock, []{ return !taskQueue.empty(); });  // 等待+条件检查
        int task = taskQueue.front();
        taskQueue.pop();
        lock.unlock();
        process(task);
    }
}
默写检查点：①lock_guard用mutex，unique_lock配合cv ②cv.wait第二个参数是lambda判断条件 ③生产者push后notify，消费者wait前检查条件
11.3 简易线程池模板后端
class ThreadPool {
    vector<thread> workers;
    queue<function<void()>> tasks;
    mutex mtx;
    condition_variable cv;
    bool stop = false;

public:
    ThreadPool(int numThreads) {
        for (int i = 0; i < numThreads; i++) {
            workers.emplace_back([this] {
                while (true) {
                    function<void()> task;
                    {
                        unique_lock<mutex> lock(mtx);
                        cv.wait(lock, [this] { return stop || !tasks.empty(); });
                        if (stop && tasks.empty()) return;
                        task = move(tasks.front());
                        tasks.pop();
                    }
                    task();  // 执行任务（不加锁）
                }
            });
        }
    }

    template<class F>
    void submit(F&& f) {
        {
            lock_guard<mutex> lock(mtx);
            tasks.emplace(forward<F>(f));
        }
        cv.notify_one();
    }

    ~ThreadPool() {
        {
            lock_guard<mutex> lock(mtx);
            stop = true;
        }
        cv.notify_all();
        for (auto& w : workers) w.join();
    }
};
默写检查点：①worker线程用lambda捕获this ②wait条件是stop||!empty ③stop后要执行完队列中剩余任务再退出 ④析构时notify_all+join

// TCP服务端（C语言Socket API）
int server() {
    int listenfd = socket(AF_INET, SOCK_STREAM, 0);

    struct sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(8080);

    bind(listenfd, (struct sockaddr*)&addr, sizeof(addr));
    listen(listenfd, 5);  // backlog=5

    while (true) {
        struct sockaddr_in clientAddr;
        int addrlen = sizeof(clientAddr);
        int connfd = accept(listenfd, (struct sockaddr*)&clientAddr, &addrlen);

        char buf[1024];
        int n = recv(connfd, buf, sizeof(buf), 0);
        buf[n] = '\0';
        send(connfd, buf, n, 0);  // echo回显
        close(connfd);
    }
}

// TCP客户端
int client() {
    int sockfd = socket(AF_INET, SOCK_STREAM, 0);

    struct sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);

    connect(sockfd, (struct sockaddr*)&addr, sizeof(addr));

    const char* msg = "Hello Server";
    send(sockfd, msg, strlen(msg), 0);

    char buf[1024];
    int n = recv(sockfd, buf, sizeof(buf), 0);
    buf[n] = '\0';
    close(sockfd);
}
默写检查点：①服务端流程：socket→bind→listen→accept→recv/send→close ②客户端流程：socket→connect→send/recv→close ③htons转换端口字节序 ④recv返回的是字节数，手动加'\0'

12.1 矩阵基础运算线性代数扩展
using Matrix = vector<vector<int>>;

// 矩阵加法 A + B
Matrix add(Matrix& A, Matrix& B) {
    int m = A.size(), n = A[0].size();
    Matrix C(m, vector<int>(n));
    for (int i = 0; i < m; i++)
        for (int j = 0; j < n; j++)
            C[i][j] = A[i][j] + B[i][j];
    return C;
}

// 矩阵乘法 A(m×k) * B(k×n) = C(m×n)
Matrix multiply(Matrix& A, Matrix& B) {
    int m = A.size(), k = A[0].size(), n = B[0].size();
    Matrix C(m, vector<int>(n, 0));
    for (int i = 0; i < m; i++)        // C的行
        for (int j = 0; j < n; j++)    // C的列
            for (int p = 0; p < k; p++)  // A的列=B的行
                C[i][j] += A[i][p] * B[p][j];
    return C;
}

// 矩阵快速幂（求A^n）— 用于加速递推
// 例：斐波那契数列可以用矩阵快速幂在O(logN)内求第n项
Matrix matPow(Matrix A, int n) {
    int sz = A.size();
    Matrix result(sz, vector<int>(sz, 0));
    // 初始化为单位矩阵
    for (int i = 0; i < sz; i++) result[i][i] = 1;

    while (n > 0) {
        if (n & 1) result = multiply(result, A);
        A = multiply(A, A);
        n >>= 1;
    }
    return result;
}

// 前缀和矩阵（区域和查询O(1)）
// prefix[i][j] = (0,0)到(i-1,j-1)的元素总和
vector<vector<int>> buildPrefix(Matrix& a) {
    int m = a.size(), n = a[0].size();
    vector<vector<int>> pre(m + 1, vector<int>(n + 1, 0));
    for (int i = 1; i <= m; i++)
        for (int j = 1; j <= n; j++)
            pre[i][j] = a[i-1][j-1] + pre[i-1][j] + pre[i][j-1] - pre[i-1][j-1];
    return pre;
}
// 查询(x1,y1)到(x2,y2)的区域和
int query(vector<vector<int>>& pre, int x1, int y1, int x2, int y2) {
    return pre[x2+1][y2+1] - pre[x1][y2+1] - pre[x2+1][y1] + pre[x1][y1];
}
默写检查点：①矩阵乘法三层循环顺序是ijk（ijk可交换但缓存效率不同）②快速幂初始化为单位矩阵 ③前缀和公式含容斥原理：右下-左下-右上+左上
